\begin{frame}{Актуальность задачи}
\setlstep{0.3cm}{0cm}{0cm}
\begin{itemize}
    \item Медицинские данные зачастую являются персональной информацией пациента, поэтому их централизованное хранение невозможно.
    
    \item Коммуникации между клиентами и сервером являются узким местом, поэтому важно уметь производить коммуникации эффективно
\end{itemize}

\begin{frame}[fragile]{Постановка задачи}

\setlstep{0.3cm}{0cm}{0cm}
\begin{itemize}
	% \item Федеративное обучение сводится к решению следующей оптимизационной задачи
 % \begin{equation}\label{eq0:fl_setup}
 %     \min_{\x \in \bR^d}\left\{f(\x) \overset{\mathrm{def}}{=} \dfrac{1}{M}\sum_{m=1}^M f_m(\x)\right\},
 % \end{equation}
 % где $f_m(\x) \overset{\mathrm{def}}{=} \bE_{\z_m \sim \cc{D}_m}[f_m(\x, \z_m)]$ -- функция потерь клиента $m$, $\cc{D}_m$ -- распределение данных клиента $m$, $\x$ -- параметры модели.

        \item Цель нашего исследования научится производить коммуникации эффективно за счет уменьшения количества клиентов, которые учавствуют в коммуникации.

        \item Работая с меньшим количеством клиентов, мы уменьшаем нагрузку на сеть, что позволяет экономить ресурсы, необходимые для передачи данных между клиентами и сервером, например, деньги.

\end{itemize}
\end{frame}

\begin{frame}{FedAvg \footcite{mcmahan2017communication}}
\begin{minipage}{0.65\textwidth}
\begin{algorithm}[H]
\caption{Federated Averaging}
\begin{algorithmic}[1]
\State \textbf{Initialize} $\x_0$
\For {each round $t = 1, 2, \dots$}
    \State $S_t \gets$ (случайное множество $m$ клиентов)
    \For {клиента $k \in S_t$ \textbf{параллельно}}
        \State $\x_{t+1}^k \gets \textsc{Обновление Клиента}(k, \x_t)$
    \EndFor
    \State $m_t \gets \sum_{k \in S_t} n_k$
    \State $\x_{t+1} \gets \sum_{k \in S_t} \frac{n_k}{m_t} \x_{t+1}^k$
\EndFor
\end{algorithmic}
\end{algorithm}
\end{minipage}

\begin{frame}{FedAvg, локальные вычисления клиентов\footcite{mcmahan2017communication}}
\begin{minipage}{0.65\textwidth}
\caption{Federated Averaging}
\begin{algorithmic}[1]
\begin{algorithm}[H]
\caption{Обновление Клиента($k$, $\x$)}
\begin{algorithmic}[1]
\State $\mathcal{B} \gets$ (split $\mathcal{P}_k$ into batches of size $B$)
\For {each local epoch $i$ from $1$ to $E$}
    \For {batch $b \in \mathcal{B}$}
        \State $\x \gets \x - \eta \nabla \ell(w; b)$
    \EndFor
\EndFor
\State \Return $\x$ to server
\end{algorithmic}
\end{algorithm}
\end{algorithmic}
