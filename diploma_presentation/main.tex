\begin{frame}{Постановка задачи}
\small
\begin{itemize}
    \item Решаются задачи численного оптимального управления и численного стохастического оптимального управления в дискретном времени, где С, L  - переменные управления, а K и z - переменные состояния.
    \item Требуется численно исследовать модель реального делового цикла для закрытой экономики в двух постановках --- детерминированной и стохастической.
\end{itemize}
\end{frame}

\begin{frame}{Описание детерминированной модели}
    \scriptsize

    \begin{itemize}
        \item Агенты: репрезентативное домохозяйство, репрезентативная фирма.
        \item Задача домохозяйства - максимизация фукнции полезности:
    \end{itemize}

    \[
    U = \max_{C,L,K}
    \mathbb{E} \sum_{t=0}^{\infty}\beta^t
    \left(
    \frac{C_t^{1-\theta}-1}{1-\theta}-\varphi \frac{L_t^{1+\psi}}{1+\psi}
    \right)
    \]

    При условии соблюдения балансового ограничения и с учетом правила накопления активов, которые объединяют потребление, инвестиции и доходы от труда и капитала:
    \[
    K_{t+1}=(1-\delta)K_t + W_tL_t+R_tK_t - C_t
    \]

    \begin{itemize}
        \item Задача фирмы - максимизация прибыли:
    \end{itemize}
    \[
        Pr = \max_{C,L,K} \left\{AK_t^{\alpha}L_t^{1-\alpha} - W_tL_t - R_tK_t\right\}
    \]
\end{frame}

\begin{frame}{Вывод модели: условия первого порядка (FOC)}
\scriptsize
\begin{itemize}
    \item В условиях совершенной конкуренции задача эквивалентна задаче социального планировщика (первая теорема благосостояния).
    \item Лагранжиан задачи домохозяйства с множителем $\lambda_t$:
\end{itemize}
\[
\mathcal{L}=E_0\sum_{t=0}^{\infty}\beta^t\Big[u(C_t,L_t)+\lambda_t\big((1-\delta)K_t+W_tL_t+R_tK_t-C_t-K_{t+1}\big)\Big].
\]
\begin{itemize}
    \item Условия первого порядка по $C_t,\,L_t,\,K_{t+1}$ дают $\lambda_t=u_C(C_t,L_t)$, внутрипериодный выбор труда и межвременное уравнение Эйлера:
\end{itemize}
\[
-\frac{u_L(C_t,L_t)}{u_C(C_t,L_t)}=W_t,
\qquad
C_t^{-\sigma}=\beta\,E_t\big[C_{t+1}^{-\sigma}(1-\delta+R_{t+1})\big].
\]
\begin{itemize}
    \item FOC фирмы (предельные продукты факторов):
    \(\; W_t=(1-\alpha)\dfrac{Y_t}{L_t},\quad R_t=\alpha\dfrac{Y_t}{K_t}.\)
\end{itemize}
\end{frame}


\begin{frame}{Итоговая модель: динамическая система}
\scriptsize
\begin{itemize}
    \item Труд выражается явно через $K_t$ и $C_t$, поэтому все эндогенные переменные становятся функциями пары $(K_t,C_t)$. Итоговая система:
\end{itemize}
\[
\begin{cases}
L_t = \left( \dfrac{(1-\alpha)A}{\phi}\cdot\dfrac{K_t^{\alpha}}{C_t} \right)^{\frac{1}{\psi+\alpha}}, \\[8pt]
Y_t = A K_t^{\alpha} L_t^{1-\alpha},\quad
R_t = \alpha \dfrac{Y_t}{K_t},\quad
w_t = (1-\alpha)\dfrac{Y_t}{L_t},\quad
I_t = Y_t - C_t, \\[8pt]
K_{t+1} = Y_t - C_t + (1-\delta) K_t, \\[6pt]
C_{t+1} = \beta\big(1 + R(K_{t+1},C_{t+1}) - \delta\big)\,C_t.
\end{cases}
\]
\begin{itemize}
    \item Переход по капиталу $K_{t+1}$ --- явный; переход по потреблению $C_{t+1}$ задан неявно (уравнение Эйлера решается методом Ньютона).
    \item $K_0$ задано, начальное $C_0$ подбирается методом стрельбы так, чтобы траектория сходилась к стационарному состоянию.
\end{itemize}
\end{frame}



\begin{frame}{Описание детерминированной модели}
\scriptsize


    \begin{itemize}
        \item Стационарное состояние (SS):
        \(K_t=K_{t+1}=K^*,\ C_t=C_{t+1}=C^*,\ L_t=L_{t+1}=L^*\).
    \end{itemize}
    \begin{align*}
        R^* &=\frac{1}{\beta}-1+\delta\\
        L^* &=\left(\frac{(1-\alpha)R^*}{\alpha \phi v}\right)^{\frac{1}{\psi+1}}\\
        K^* &=\left(\frac{\alpha A (L^*)^{1-\alpha}}{R^*}\right)^{\frac{1}{1-\alpha}}\\
        C^* &=\frac{\frac{1}{\beta}-1+\delta(1-\alpha)}{\alpha}K^*\\
    \end{align*}

    \begin{itemize}
        % \item Численное решение: лог-линеаризация системы FOC; метод стрельбы для подбора начального потребления \(C_0\); построение фазовых портретов и временных рядов.
        \item Параметры: \(\beta=0.96,\ \alpha=0.33,\ \delta=0.1,\ \sigma=1,\ \phi=0.75,\ \psi=0.35\).
    \end{itemize}
\end{frame}

\begin{frame}{Обзор решений: Глобальные методы решения RBC}
    \scriptsize
    % В отличие от локальных, глобальные методы не линеаризуют модель и ищут численное решение на всем пространстве состояний. Это важно для больших шоков, нелинейностей и ограничений.

    \vspace{0.25em}
    \begin{itemize}
        \item \textbf{Метод Стрельбы (shooting):} подбор начального \(C_0\), чтобы траектория по уравнению Эйлера пришла в SS; в моей реализации используется бисекция с сужением интервала по направлению движения \(K\) и \(C\).
        % \item \textbf{Метод проекций (Projection / Weighted Residuals):} решение аппроксимируется, например, полиномами Чебышева; коэффициенты выбираются так, чтобы минимизировать невязку во всей области. Обычно точнее пертурбаций для нелинейных моделей.
        % \item \textbf{Итерация по функции ценности (VFI):} итеративное решение уравнения Беллмана для \(V(K)\); естественно учитывает ограничения типа \(K_{t+1}\ge 0\) и сложные нелинейности.
    \end{itemize}
\end{frame}

\begin{frame}{Стохастическая DSGE-модель}
    \scriptsize
    \begin{itemize}
        \item Переход от детерминированной RBC к DSGE: добавляем случайный технологический шок в производственную функцию:
        \[
        Y_t = A e^{z_t} K_t^{\alpha} L_t^{1-\alpha}.
        \]
        \item Динамика шока TFP задается AR(1)-процессом:
        \[
        z_t = \rho z_{t-1} + \varepsilon_t,\qquad \varepsilon_t \sim \mathcal{N}(0,\sigma_z^2),\quad |\rho|<1.
        \]
        \item Остальные условия (Эйлер, внутрипериодный выбор труда, закон движения капитала) остаются как в детерминированной модели.
        \item В DSGE решением является распределение траекторий макропеременных, а не одна детерминированная траектория.
    \end{itemize}
\end{frame}

\begin{frame}{Выводы}
\small
\begin{itemize}
    \item Реализовано численное решение детерминированной RBC-модели: выписаны условия первого порядка и уравнение Эйлера, найдено стационарное состояние, система сведена к динамике пары $(K_t, C_t)$.

    \item Построенные фазовые и 3D-траектории сходятся к стационарному состоянию при $K_0$ как ниже, так и выше равновесия --- это подтверждает седловую устойчивость и корректность алгоритма.
    \item Сравнение локального и глобального решений: линеаризация точна лишь вблизи стационара, тогда как при больших шоках глобальный подход содержательнее (но вычислительно дороже).
\end{itemize}
